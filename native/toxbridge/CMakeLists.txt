cmake_minimum_required(VERSION 3.22)
project(tochat LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(SODIUM_DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${ROOT}/vendor/libsodium-cmake" sodium)
# Use the exact upstream core source manifest, without building AV, groups UI,
# bootstrap daemons or relying on host pkg-config during Android cross builds.
file(READ "${ROOT}/vendor/c-toxcore/CMakeLists.txt" TOX_MANIFEST)
string(REGEX MATCH "set\\(toxcore_SOURCES[^)]*\\)" TOX_BLOCK "${TOX_MANIFEST}")
string(REGEX MATCHALL "[a-zA-Z0-9_/.-]+\\.c" TOX_SOURCES "${TOX_BLOCK}")
list(TRANSFORM TOX_SOURCES PREPEND "${ROOT}/vendor/c-toxcore/")
add_library(toxcore STATIC ${TOX_SOURCES} "${ROOT}/vendor/c-toxcore/toxencryptsave/toxencryptsave.c")
target_include_directories(toxcore PUBLIC "${ROOT}/vendor/c-toxcore")
target_compile_definitions(toxcore PRIVATE CMP_NO_FLOAT=1 MIN_LOGGER_LEVEL=LOGGER_LEVEL_WARNING)
target_link_libraries(toxcore PUBLIC sodium)
if(WIN32)
  target_link_libraries(toxcore PUBLIC ws2_32 iphlpapi winpthread)
endif()
add_library(toxbridge SHARED bridge.cpp)
target_include_directories(toxbridge PRIVATE "${ROOT}/vendor")
target_link_libraries(toxbridge PRIVATE toxcore)
if(ANDROID)
  target_sources(toxbridge PRIVATE android_jni.cpp)
  target_link_libraries(toxbridge PRIVATE log)
  target_link_options(toxbridge PRIVATE "-Wl,-z,max-page-size=16384")
else()
  target_link_options(toxbridge PRIVATE -static-libstdc++ -static-libgcc)
endif()
